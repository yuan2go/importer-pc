using System;
using System.Collections.Generic;
using System.Diagnostics;
using System.IO;
using System.Linq;
using System.Runtime.InteropServices;
using System.Security.Cryptography;
using System.Text;
using System.Text.Json;
using System.Threading.Tasks;
using System.Windows.Automation;

internal static class Program
{
    static string output = "";
    static readonly List<string> lines = [];
    static int checks;
    static void Check(bool value, string message) { if (!value) throw new InvalidOperationException(message); checks++; lines.Add("PASS " + message); }
    [STAThread] static int Main(string[] args)
    {
        output = Path.GetFullPath(args[1]); Directory.CreateDirectory(output);
        try { Verify(Path.GetFullPath(args[0]), args[2]).GetAwaiter().GetResult(); File.WriteAllText(Path.Combine(output, "RESULT.json"), JsonSerializer.Serialize(new { Status = "PASS", SourceCommit = args[2], Assertions = checks, Scope = "Actual standalone EXE: welcome, native preparation, cancellation, original setup handoff and close; no installation or device operation" }, new JsonSerializerOptions { WriteIndented = true })); return 0; }
        catch (Exception error) { lines.Add("FAIL " + error); return 1; }
        finally { File.WriteAllLines(Path.Combine(output, "NATIVE.log"), lines); }
    }
    static Process Launch(string executable)
    {
        var start = new ProcessStartInfo(executable) { UseShellExecute = false, WorkingDirectory = Path.GetDirectoryName(executable)!, WindowStyle = ProcessWindowStyle.Hidden };
        start.Environment["DOTNET_ROOT"] = Path.Combine(output, "no-shared-runtime");
        start.Environment["DOTNET_ROOT_X64"] = start.Environment["DOTNET_ROOT"];
        start.Environment["DOTNET_MULTILEVEL_LOOKUP"] = "0";
        var process = Process.Start(start) ?? throw new IOException("Cannot launch actual installer");
        _ = process.Handle; return process;
    }
    static HashSet<string> Workspaces() => Directory.EnumerateDirectories(Path.GetTempPath(), "ipc-installer-*").ToHashSet(StringComparer.OrdinalIgnoreCase);
    static AutomationElement ById(IntPtr window, string id) => AutomationElement.FromHandle(window).FindFirst(TreeScope.Descendants, new PropertyCondition(AutomationElement.AutomationIdProperty, id)) ?? throw new InvalidOperationException("Missing control " + id);
    static AutomationElement? Button(IntPtr window, string name) => AutomationElement.FromHandle(window).FindFirst(TreeScope.Descendants, new AndCondition(new PropertyCondition(AutomationElement.ControlTypeProperty, ControlType.Button), new PropertyCondition(AutomationElement.NameProperty, name)));
    static void Invoke(AutomationElement element) => ((InvokePattern)element.GetCurrentPattern(InvokePattern.Pattern)).Invoke();
    static async Task<IntPtr> WaitWindow(int processId, string title)
    {
        for (var i = 0; i < 300; i++) { var handle = Native.Find(processId, title); if (handle != IntPtr.Zero) return handle; await Task.Delay(50); }
        throw new TimeoutException("Missing native window: " + title);
    }
    static async Task Exit(Process process)
    {
        await process.WaitForExitAsync().WaitAsync(TimeSpan.FromSeconds(20));
        Check(process.ExitCode == 0, "actual process exits normally with code zero");
    }
    static async Task Verify(string original, string source)
    {
        var drop = Path.Combine(output, "field-drop"); Directory.CreateDirectory(drop);
        var executable = Path.Combine(drop, "菜鸟供包台安装程序.exe"); File.Copy(original, executable, false);
        var info = FileVersionInfo.GetVersionInfo(executable);
        Check(info.ProductName == "菜鸟供包台安装程序" && info.FileDescription == "菜鸟供包台安装程序", "standalone EXE has the intended product and file-description names");
        Check(Directory.GetFiles(drop).Length == 1 && !Directory.GetDirectories(drop).Any(), "field delivery contains exactly one EXE without sidecars");
        var before = Workspaces();
        using (var process = Launch(executable))
        {
            var window = await WaitWindow(process.Id, "菜鸟供包台安装程序");
            Check(ById(window, "Heading").Current.Name == "欢迎安装菜鸟供包台" && ById(window, "Next").Current.IsEnabled, "actual single file opens the native welcome page without a shared runtime root");
            process.Refresh();
            lines.Add("Modules=" + string.Join(",", process.Modules.Cast<ProcessModule>().Select(module => module.ModuleName)));
            var runtime = process.Modules.Cast<ProcessModule>().Single(module => module.ModuleName.Equals("System.Private.CoreLib.dll", StringComparison.OrdinalIgnoreCase)).FileName;
            lines.Add("BundledRuntime=" + runtime);
            Check(!runtime.StartsWith("D:\\tools\\dotnet\\shared", StringComparison.OrdinalIgnoreCase), "runtime core library is loaded from the bundled runtime rather than the developer SDK shared framework");
            Native.Capture(window, Path.Combine(output, "welcome.png"));
            Native.Resize(window, 640, 420); await Task.Delay(250); Native.Capture(window, Path.Combine(output, "welcome-small.png"));
            var bounds = ById(window, "Cancel").Current.BoundingRectangle;
            var whole = AutomationElement.FromHandle(window).Current.BoundingRectangle;
            Check(bounds.Bottom <= whole.Bottom && bounds.Right <= whole.Right && bounds.Width >= 100, "cancel action remains inside the small native window");
            Invoke(ById(window, "Cancel")); await Exit(process);
        }
        Check(Workspaces().SetEquals(before), "cancelling welcome creates no preparation workspace");
        using (var process = Launch(executable))
        {
            var window = await WaitWindow(process.Id, "菜鸟供包台安装程序");
            Invoke(ById(window, "Next"));
            Invoke(ById(window, "Cancel"));
            await Exit(process);
        }
        Check(Workspaces().SetEquals(before), "cancelling actual preparation cleans the private workspace and does not open setup");
        using (var process = Launch(executable))
        {
            var window = await WaitWindow(process.Id, "菜鸟供包台安装程序"); Invoke(ById(window, "Next"));
            IntPtr setup = IntPtr.Zero; Process? child = null; string? media = null; var progressSeen = false;
            for (var i = 0; i < 1200; i++)
            {
                process.Refresh(); if (process.HasExited) throw new IOException("Installer exited before setup handoff");
                var roots = Workspaces().Except(before, StringComparer.OrdinalIgnoreCase).ToArray();
                if (roots.Length == 1)
                {
                    media = Path.Combine(roots[0], "media");
                    foreach (var candidate in Process.GetProcessesByName("Ipc.Desktop"))
                    {
                        if (candidate.MainModule?.FileName == Path.Combine(media, "desktop", "Ipc.Desktop.exe"))
                        {
                            var handle = Native.Find(candidate.Id, "安装与部署向导 · 菜鸟供包台");
                            if (handle != IntPtr.Zero) { child = candidate; setup = handle; break; }
                        }
                        candidate.Dispose();
                    }
                }
                if (setup != IntPtr.Zero) break;
                if (!progressSeen)
                {
                    var range = (RangeValuePattern)ById(window, "Progress").GetCurrentPattern(RangeValuePattern.Pattern);
                    if (range.Current.Value is > 0 and < 100)
                    {
                        progressSeen = true; lines.Add("ActualProgress=" + range.Current.Value);
                        Native.Capture(window, Path.Combine(output, "preparing.png"));
                    }
                }
                await Task.Delay(25);
            }
            Check(setup != IntPtr.Zero && child is not null && media is not null, "actual single-file payload opens the original native deployment wizard");
            Check(progressSeen, "actual byte-based preparation progress is exposed through native automation");
            using (child!)
            {
                using var release = JsonDocument.Parse(File.ReadAllText(Path.Combine(media!, "app", "application-release.json")));
                Check(release.RootElement.GetProperty("SourceCommit").GetString() == source, "expanded payload retains the exact frozen source identity");
                Native.Capture(setup, Path.Combine(output, "setup-first.png"));
                Invoke(Button(setup, "检查安装媒介") ?? throw new InvalidOperationException("Missing media-review action"));
                for (var i = 0; i < 300 && Button(setup, "检查部署并查看摘要") is null; i++) await Task.Delay(50);
                Check(Button(setup, "检查部署并查看摘要") is { Current.IsEnabled: true }, "the original wizard accepts the embedded inventory and reaches directory/account selection");
                Native.Capture(setup, Path.Combine(output, "setup-directories.png"));
                Native.Close(setup); await Exit(child!);
            }
            await Exit(process);
        }
        Check(Workspaces().SetEquals(before), "closing the original setup cleans this run's extracted medium");
        Check(Directory.GetFiles(drop).Length == 1, "the external delivery remains one EXE after all runs");
        lines.Add("NOT_RUN actual UAC installation, upgrade/uninstall, devices, SDK, RCS and field acceptance.");
    }
}

internal static class Native
{
    delegate bool EnumCallback(IntPtr window, IntPtr data);
    [DllImport("user32.dll")] static extern bool EnumWindows(EnumCallback callback, IntPtr data);
    [DllImport("user32.dll")] static extern uint GetWindowThreadProcessId(IntPtr window, out uint process);
    [DllImport("user32.dll", CharSet = CharSet.Unicode)] static extern int GetWindowText(IntPtr window, StringBuilder text, int count);
    [DllImport("user32.dll")] static extern bool PostMessage(IntPtr window, uint message, IntPtr wParam, IntPtr lParam);
    [DllImport("user32.dll")] static extern bool GetWindowRect(IntPtr window, out Rect rectangle);
    [DllImport("user32.dll")] static extern bool PrintWindow(IntPtr window, IntPtr context, uint flags);
    [DllImport("user32.dll")] static extern bool SetWindowPos(IntPtr window, IntPtr after, int x, int y, int width, int height, uint flags);
    [StructLayout(LayoutKind.Sequential)] struct Rect { public int Left, Top, Right, Bottom; }
    public static IntPtr Find(int processId, string title)
    {
        var found = IntPtr.Zero;
        EnumWindows((window, _) => { GetWindowThreadProcessId(window, out uint owner); if (owner != processId) return true; var text = new StringBuilder(1024); GetWindowText(window, text, text.Capacity); if (text.ToString() != title) return true; found = window; return false; }, IntPtr.Zero);
        return found;
    }
    public static void Close(IntPtr window) { if (!PostMessage(window, 0x0010, IntPtr.Zero, IntPtr.Zero)) throw new IOException("Cannot close own test window"); }
    public static void Resize(IntPtr window, int width, int height) => SetWindowPos(window, IntPtr.Zero, 0, 0, width, height, 0x0002 | 0x0004);
    public static void Capture(IntPtr window, string file)
    {
        if (!GetWindowRect(window, out var rectangle)) throw new IOException("Cannot read own window bounds");
        using var bitmap = new System.Drawing.Bitmap(rectangle.Right - rectangle.Left, rectangle.Bottom - rectangle.Top);
        using var graphics = System.Drawing.Graphics.FromImage(bitmap); var context = graphics.GetHdc();
        try { if (!PrintWindow(window, context, 2)) throw new IOException("Cannot render own installer HWND"); }
        finally { graphics.ReleaseHdc(context); }
        bitmap.Save(file, System.Drawing.Imaging.ImageFormat.Png);
    }
}
