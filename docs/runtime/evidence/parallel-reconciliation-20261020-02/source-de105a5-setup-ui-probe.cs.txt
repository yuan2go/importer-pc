    [STAThread] static int Main(string[] args)
    {
        var output = Path.GetFullPath(args[1]); Directory.CreateDirectory(output); var lines = new List<string>();

        app.Startup += async (_, _) =>
        {
            try { await Verify(args[0], output, lines); app.Shutdown(0); }
            catch (Exception e) { lines.Add("FAIL " + e); app.Shutdown(1); }
            finally { File.WriteAllLines(Path.Combine(output, "UI.log"), lines); }
        };
        return app.Run();
    }
    static IEnumerable<T> All<T>(DependencyObject parent) where T : DependencyObject
    {
        for (var i = 0; i < VisualTreeHelper.GetChildrenCount(parent); i++) { var child = VisualTreeHelper.GetChild(parent, i); if (child is T value) yield return value; foreach (var nested in All<T>(child)) yield return nested; }
    }
    static async Task Verify(string mediaRoot, string output, List<string> log)
    {
        void Check(bool yes, string name) { if (!yes) throw new Exception(name); log.Add("PASS " + name); }
        var type = System.Reflection.Assembly.Load("Ipc.Desktop").GetType("Ipc.Desktop.StartupSetupWindow")!;
        var window = (Window)Activator.CreateInstance(type, new BusinessLaunch(Path.Combine(mediaRoot, "app", "config"), "D:\\SYNTHETIC\\data", null, null))!;
        void Set(string name, object value) => type.GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.SetValue(window, value);
        T Get<T>(string name) => (T)type.GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(window)!;
        void Render() => type.GetMethod("Render", BindingFlags.Instance | BindingFlags.NonPublic | BindingFlags.DeclaredOnly)!.Invoke(window, null);
        void Click(string name) => All<Button>(window).Single(b => Equals(b.Content, name)).RaiseEvent(new RoutedEventArgs(Button.ClickEvent));

        Set("enteredAccount", "SYNTHETIC\\operator"); Set("newStation", "synthetic-station"); Render(); await Task.Delay(200);
        Check(Get<TextBox>("program").Text == "D:\\SYNTHETIC\\Programs" && All<CheckBox>(window).Count(b => b.IsVisible && (Equals(b.Content, "在开始菜单创建快捷方式") || Equals(b.Content, "在桌面创建快捷方式"))) == 2, "directory and both shortcut choices visible without expanding advanced settings");
        Check(!Get<TextBox>("account").IsVisible && All<Button>(window).Any(b => Equals(b.Content, "浏览…") && b.IsVisible), "advanced account hidden while native directory browse remains visible");
        Get<CheckBox>("startMenu").IsChecked = false; Get<CheckBox>("desktopIcon").IsChecked = true; Get<TextBox>("program").Text = "D:\\SYNTHETIC\\Chosen";
        type.GetMethod("CaptureInputs", BindingFlags.Instance | BindingFlags.NonPublic)!.Invoke(window, null);
        var sid = WindowsIdentity.GetCurrent().User!.Value;
        var request = new WindowsDeploymentRequest(mediaRoot, "sportmaster", new string('a', 40), sid, "D:\\SYNTHETIC\\data", "D:\\SYNTHETIC\\config", RunMode.Simulation, null, null, [], ProgramRoot: "D:\\SYNTHETIC\\Chosen", Shortcuts: new(false, true));
        Set("reviewed", request); Set("step", 2); Render(); await Task.Delay(100);
        Check(All<TextBlock>(window).Any(t => t.Text.Contains("开始菜单快捷方式：不创建") && t.Text.Contains("桌面快捷方式：创建")), "summary displays chosen shortcut flags");

        Check(Get<TextBox>("program").Text == request.ProgramRoot && Get<CheckBox>("startMenu").IsChecked == false && Get<CheckBox>("desktopIcon").IsChecked == true, "back navigation preserves directory and both selections");

    static void Capture(Window window, string path)
    {
        window.UpdateLayout(); var bitmap = new RenderTargetBitmap((int)window.ActualWidth, (int)window.ActualHeight, 96, 96, PixelFormats.Pbgra32); bitmap.Render(window);
        var png = new PngBitmapEncoder(); png.Frames.Add(BitmapFrame.Create(bitmap)); using var file = File.Create(path); png.Save(file);
    }
