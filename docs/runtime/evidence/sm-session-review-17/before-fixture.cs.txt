using System.Diagnostics;
using System.Reflection;
using System.Threading.Channels;
using System.Windows.Threading;
using Ipc.Application.Contracts;
using Ipc.Application.Sportmaster.Desktop;
using Ipc.Platform;

static partial class Program
{
    static async Task VerifyPresentationOrderingUi(string root, string evidence, List<string> lines)
    {
        void Check(bool value, string text) { if (!value) throw new InvalidOperationException(text); lines.Add("PASS PRESENTATION-ORDER " + text); }
        async Task Until(Func<bool> ready, string step)
        { for (var i = 0; i < 200; i++) { await Dispatcher.Yield(DispatcherPriority.ApplicationIdle); if (ready()) return; await Task.Delay(20); } throw new TimeoutException(step); }
        T? Field<T>(object target, string name) => (T?)target.GetType().GetField(name, BindingFlags.Instance | BindingFlags.NonPublic)!.GetValue(target);
        await using var h = new CameraRoundHarness(root, Path.Combine(evidence, "ordering")); await h.StartAsync();
        h.Api(h.Left).Enqueue(CameraRoundHarness.Barcode("ORDER-0001"), 1);
        await CameraRoundHarness.Until(async () => (await h.App.SnapshotAsync()).Operations.Any(o => o.State == ParcelState.SlotResolved));
        var app = DispatchProxy.Create<IPresentationOrderingApplication, PresentationOrderingProxy>();
        var proxy = (PresentationOrderingProxy)(object)app; proxy.Inner = h.App;
        var window = new SupplyStationWindow(app, null, "standard"); window.Show();
        try
        {
            await SetClient(window, 1366, 768); await Until(() => window.ViewModel.Fresh && window.ViewModel.Left.ShowConfirmedTask, "initial actual snapshot");
            var basis = window.ViewModel.Snapshot!;
            ParcelSnapshot Project(string slot) => basis with { PublishedTicks = Stopwatch.GetTimestamp(),
                Operations = basis.Operations.Select(o => o.PositionId == "left" ? o with { Slot = slot } : o).ToArray() };
            ParcelPresentationUpdate Update(long version, ParcelSnapshot? snapshot, string? failure = null) =>
                new(basis.RuntimeId, version, snapshot, [], [], failure);
            // Only the display slot is projected. All commands remain real, and this
            // test never writes these projections back as Application/business facts.
            var first = Project("SLOW-ONE"); proxy.HoldNextStatus = true; proxy.Sub.Publish(Update(1, first));
            await Until(() => proxy.HeldStatus is not null, "slow first snapshot application");
            var latest = Project("SLOW-LATEST"); proxy.Sub.Publish(Update(2, latest));
            await Until(() => Field<ParcelPresentationUpdate>(window, "snapshotWaiting")?.Version == 2, "deferred latest full snapshot");
            var status = await h.App.ManagementModeStatusAsync();
            using (window.Dispatcher.DisableProcessing())
            {
                var input = basis.Inputs!.Single(i => i.PositionId == "left");
                var cue = new ParcelReadFeedback(basis.RuntimeId, "left", input.Observation.Id, input.Observation.DeviceId,
                    input.Observation.Epoch, "ORDER-0001", Stopwatch.GetTimestamp(), input.Observation.ObservedAt);
                proxy.Sub.Publish(Update(3, null) with { Reads = [cue] });
                var pending = false;
                for (var i = 0; i < 1000; i++)
                { if (Field<ParcelPresentationUpdate>(window, "pendingPresentation")?.Version == 3) { pending = true; break; } Thread.Sleep(1); }
                Check(pending, "newer read notification is pending while the older full snapshot application finishes");
                proxy.HeldStatus!.TrySetResult(status);
            }
            await Until(() => window.ViewModel.Left.DisplaySlot == "SLOW-LATEST", "newer read notification lost the deferred full snapshot");
            Check(window.ViewModel.Left.ShowConfirmedTask, "latest full result survives a newer read-only update without waiting for a watchdog publication");

            proxy.HoldNextStatus = true; proxy.HeldStatus = null;
            proxy.Sub.Publish(Update(4, Project("MUST-NOT-REVIVE")));
            await Until(() => proxy.HeldStatus is not null, "second slow snapshot application");
            proxy.Sub.Publish(Update(5, null, "Synthetic snapshot storage failure"));
            await Until(() => !window.ViewModel.Fresh && window.ViewModel.ReadStatus.Contains("Synthetic snapshot storage failure"), "immediate failure projection");
            proxy.HeldStatus!.TrySetResult(status);
            await Until(() => Field<Task>(window, "presentationSnapshotApply")?.IsCompleted == true, "late failed snapshot application completion");
            Check(!window.ViewModel.Fresh && !window.ViewModel.Left.DisplaySlotVisible,
                "an older in-flight success cannot clear a newer failure or resurrect its destination");
            proxy.Sub.Publish(Update(6, Project("RECOVERED")));
            await Until(() => window.ViewModel.Fresh && window.ViewModel.Left.DisplaySlot == "RECOVERED", "explicit newer successful publication");
            Check(window.ViewModel.Left.ShowConfirmedTask && h.Peer.Requests.Count == 1,
                "a newer successful publication recovers the projection without repeating any original request");
            CaptureInternal(evidence, "presentation-ordering", (System.Windows.Controls.Grid)window.FindName("ClientArea"), lines);
        }
        finally { proxy.HeldStatus?.TrySetResult(await h.App.ManagementModeStatusAsync()); window.Close(); }
    }
}

public interface IPresentationOrderingApplication : IParcelApplication, IParcelPresentationSource, IWorkstationAdministration { }
public class PresentationOrderingProxy : DispatchProxy
{
    public IParcelApplication Inner = null!;
    public readonly OrderingSubscription Sub = new();
    public bool HoldNextStatus;
    public TaskCompletionSource<ManagementModeStatus>? HeldStatus;
    protected override object? Invoke(MethodInfo? target, object?[]? args)
    {
        if (target!.Name == nameof(IParcelPresentationSource.SubscribePresentation)) return Sub;
        if (target.Name == nameof(IManagementMode.ManagementModeStatusAsync) && HoldNextStatus)
        { HoldNextStatus = false; HeldStatus = new(TaskCreationOptions.RunContinuationsAsynchronously); return HeldStatus.Task; }
        return target.Invoke(Inner, args);
    }
    public sealed class OrderingSubscription : IParcelPresentationSubscription
    {
        readonly Channel<ParcelPresentationUpdate> updates = Channel.CreateBounded<ParcelPresentationUpdate>(8);
        public void Publish(ParcelPresentationUpdate value) { if (!updates.Writer.TryWrite(value)) throw new InvalidOperationException("Synthetic fixture mailbox exhausted"); }
        public ValueTask<ParcelPresentationUpdate> ReadAsync(CancellationToken token) => updates.Reader.ReadAsync(token);
        public void Dispose() => updates.Writer.TryComplete();
    }
}
